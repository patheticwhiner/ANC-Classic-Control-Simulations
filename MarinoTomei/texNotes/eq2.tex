\section*{公式推导过程}
\subsection*{(1)基本符号假设}
已知原系统:
\begin{align*}
    \dot{x} = &
    \begin{bmatrix}
        A_0 & -b_0c_c \\
        0 & R_{CE}(\theta)
    \end{bmatrix} x + 
    \begin{bmatrix}
        b_0 \\ 0
    \end{bmatrix} (u-\varepsilon)
    \\ = &
    A_c x + 
    \left(
        \left[
        \begin{array}{c|c}
        -a_0e_1^\top & -b_0e_1^\top\\ \hline
        0 & -\theta_0e_1^\top 
        \end{array}
        \right]
        - e_ve_{v+1}^\top
    \right)
     x + 
    \begin{bmatrix}
        b_0 \\ 0
    \end{bmatrix} (u-\varepsilon)
\end{align*}

其中:
\begin{align*}
    a_0 = 
    \begin{bmatrix}
        a_{v-1} & a_{v-2} & \cdots & a_0    
    \end{bmatrix}^\top
    b_0 = 
    \begin{bmatrix}
        b_{v-1} & b_{v-2} & \cdots & b_0    
    \end{bmatrix}^\top
    \theta_0 = 
    \begin{bmatrix}
        0 & \theta_{1} & 0 & \theta_{2} & \cdots & \theta_m & 0    
    \end{bmatrix}^\top
\end{align*}

而坐标变换规定为：
\[
    \zeta = T_2(\theta) x
\]

其中：
\begin{align*}
    T_2(\theta) = &
    \begin{bmatrix}
        I_{v} & 0 \\
        0 & 0
    \end{bmatrix} + 
    \left[
    \begin{array}{c|c}
    \Theta & B
    \end{array}
    \right]
    \\ = &
    \begin{bmatrix}
        I_{v} & 0 \\ 
        0 & 0
    \end{bmatrix}
    + \sum_{k=1}^{v} Z^k \theta_0 e_t^\top
    - \sum_{\ell=1}^{2m+1} Z^{\ell} b_0 e_{v+\ell}^\top
\end{align*}

\begin{align*}
    T_2 = & 
    \left[
        \begin{array}{ccccc|ccccc}
            1 & 0 & 0 & \cdots & 0 & 0 & 0 & 0 & \cdots & 0 \\
            \mathbf{\theta_0} & 1 & 0 & \ddots & 0 & -\mathbf{b_0} & 0 & 0 & \ddots & 0\\
            & \mathbf{\theta_0} & 1 & \ddots & 0 &  & -\mathbf{b_0} & 0 & \ddots & 0 \\ 
            &  & \mathbf{\theta_0} & \ddots & \vdots &  &  & -\mathbf{b_0} & \ddots & \vdots \\
            &  &  & \ddots & 1 &  &  &  & \ddots & 0 \\
            &  &  &  & \mathbf{\theta_0} &  &  &  &  & -\mathbf{b_0}\\
        \end{array}
    \right]
    \quad
    Z = \begin{bmatrix}
        0 & 0 & \cdots & 0 \\
        1 & 0 & \cdots & 0 \\
        0 & 1 & \ddots & 0 \\
        \vdots & \vdots & \ddots & \vdots \\
        0 & 0 & \cdots & 1 \\
        \vdots & \vdots &  & \vdots \\
        0 & 0 & \cdots & 0
    \end{bmatrix}\\
    a[0] =& Z^0 a_0 \quad a[1] = Z^2 a_0 + e_2 \quad a[m] = Z^{2m} a_0 + e_{2m}\\
    b[0] =& Z^0 b_0 \quad b[1] = Z^2 b_0 \qquad b[m] = Z^{2m} b_0\\
\end{align*}
注意T矩阵中参数$\boldsymbol{\theta}$分布在矩阵下三角，参数$\boldsymbol{b}$完全分布在矩阵上三角

\subsection*{(2)推导步骤示例}
首先对原系统进行坐标变换：
\[
    T_2\dot{x} =    
    T_2 A_c x + 
    T_2 \left(
        \left[
        \begin{array}{c|c}
        -a_0e_1^\top & -b_0e_1^\top\\ \hline
        0 & -\theta_0e_1^\top 
        \end{array}
        \right]
        - e_ve_{v+1}^\top
    \right)
     x + 
    T_2 \begin{bmatrix}
        b_0 \\ 0
    \end{bmatrix} (u-\varepsilon)
\]

\begin{enumerate}
    \item 替换坐标向量 $\zeta$
    \begin{align*}
        \dot{\zeta} = &   
        A_c \zeta + (T_2A_c-A_cT_2)x +
        T_2 \left(
        \left[
        \begin{array}{c|c}
        -a_0e_1^\top & -b_0e_1^\top\\ \hline
        0 & -\theta_0e_1^\top 
        \end{array}
        \right]
        x - e_ve_{v+1}^\top x + 
        \begin{bmatrix}
            b_0 \\ 0
        \end{bmatrix} 
        (u-\varepsilon)
        \right)
        \\ = &
        A_c \zeta + 
        [-{\theta_0}e_1^\top 
        +(Z^v{\theta_0}+e_v+{b_0})e_{v+1}^\top]x
        - T_2 e_ve_{v+1}^\top x
        \\ + & 
        \left( 
            \begin{bmatrix}
                I_{v} & 0 \\
                0 & 0
            \end{bmatrix} +
            \left[
            \begin{array}{c|c}
            \Theta & B 
            \end{array}
            \right]
        \right) 
        \left(
        \left[
        \begin{array}{c|c}
        -a_0e_1^\top & -b_0e_1^\top\\ \hline
        0 & -\theta_0e_1^\top 
        \end{array}
        \right]
        x + \begin{bmatrix}
            b_0 \\ 0
        \end{bmatrix} 
        (u-\varepsilon)
        \right)
        \\ = & 
        A_c \zeta + 
        [-{\theta_0}e_1^\top 
        +(Z^v{\theta_0}+e_v+{b_0})e_{v+1}^\top]x
        - (Z^v{\theta_0}+e_v)e_{v+1}^\top x
        \\ + & 
        \left(-a[0]e+b[0](u-\varepsilon)-b_0e_{v+1}^\top x\right) +
        \left[
        \begin{array}{c|c}
        \Theta & B 
        \end{array}
        \right]
        \left(
        \left[
        \begin{array}{c|c}
        -a_0e_1^\top & -b_0e_1^\top\\ \hline
        0 & -\theta_0e_1^\top 
        \end{array}
        \right]
        x + \begin{bmatrix}
            b_0 \\ 0
        \end{bmatrix} 
        (u-\varepsilon)
        \right)
        \\ = & 
        A_c \zeta -{\theta_0}e_1^\top x + 
        \left(-a[0]e+b[0](u-\varepsilon)\right) +
        \left[
        \begin{array}{c|c}
        \Theta & B 
        \end{array}
        \right]
        \left(
        \left[
        \begin{array}{c|c}
        -a_0e_1^\top & -b_0e_1^\top\\ \hline
        0 & -\theta_0e_1^\top 
        \end{array}
        \right]
        x + \begin{bmatrix}
            b_0 \\ 0
        \end{bmatrix} 
        (u-\varepsilon)
        \right)
    \end{align*}
    \item 展开等式右侧
    \begin{align*}
        & \left[
        \begin{array}{c|c}
        \Theta & B 
        \end{array}
        \right]
        \left[
        \begin{array}{c|c}
        -a_0e_1^\top & -b_0e_1^\top\\ \hline
        0 & -\theta_0e_1^\top 
        \end{array}
        \right] x
        \\ = &
            \left[
            \begin{array}{c|c}
            \Theta & 0
            \end{array}
            \right]
            \left[
            \begin{array}{c|c}
            -a_0e_1^\top & -b_0e_1^\top\\ \hline
            0 & 0
            \end{array}
            \right] x +
            \left[
            \begin{array}{c|c}
            0 & B
            \end{array}
            \right]
            \left[
            \begin{array}{c|c}
            0 & 0\\ \hline
            0 & -\theta_0e_1^\top 
            \end{array}
            \right] x
        \\ = &
            -\sum_{k=1}^{m} \theta_k Z^{2k} 
            \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} 
            (a_0e_1^\top + b_0e_{v+1}^\top)x
            + \left(\sum_{\ell=1}^{2m+1} Z^{\ell} b_0 e_{v+\ell}^\top
            \begin{bmatrix}0 \\ I_{2m+1} \end{bmatrix}\right)
            \theta_0 e_{v+1}^\top x
        \\ = &
            -\sum_{k=1}^{m} \theta_k
            (a[k]e_1^\top -e_{2k}e_1^\top + b[k]e_{v+1}^\top)x
            +\left(\sum_{\ell=1}^{2m+1} Z^{\ell} b_0 e_{\ell}^\top \right)
            \theta_0 e_{v+1}^\top x
        \\ = &
            -\sum_{k=1}^{m} \theta_k a[k]e_1^\top x
            +\sum_{k=1}^{m} \theta_k e_{2k}e_1^\top x
            -\sum_{k=1}^{m} \theta_k b[k]e_{v+1}^\top x
            +\sum_{k=1}^{m} Z^{2k} b_0 \theta_k e_{v+1}^\top x
        \\ = &
            -\sum_{k=1}^{m} \theta_k a[k]e_1^\top x + \theta_0 e_1^\top x
    \end{align*}
    \begin{align*}
        & \left[
        \begin{array}{c|c}
        \Theta & B 
        \end{array}
        \right]
        \begin{bmatrix}
            b_0 \\ 0
        \end{bmatrix} 
        (u-\varepsilon)
         = 
        \left[
        \begin{array}{c|c}
        \Theta & 0
        \end{array}
        \right]
        \begin{bmatrix}
            b_0 \\ 0
        \end{bmatrix} 
        (u-\varepsilon)
        \\ = &
        \sum_{t=1}^{v} Z^{t}\theta_0 e_t^\top 
        \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} 
        b_0 (u-\varepsilon)
         = 
        \sum_{\ell=1}^{m} \theta_{\ell}Z^{2\ell} 
        \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} 
        b_0 (u-\varepsilon)
        \\ = &
        \sum_{\ell=1}^{m} \theta_{\ell} b[\ell] (u-\varepsilon)
    \end{align*}
    \item 汇总
    \begin{align*}
        \dot{\zeta}  = & 
        A_c \zeta + \left(-a[0]e+b[0](u-\varepsilon)\right) + 
        \sum_{k=1}^{m} \theta_{k} \left(-a[k]e + b[k](u-\varepsilon)\right) 
    \end{align*}
\end{enumerate}

\subsection*{(3)简化推导步骤}
\begin{align*}
    \zeta = \left(
    \begin{bmatrix}
        I_{v} & 0 \\
        0 & 0
    \end{bmatrix} + 
    \left[
    \begin{array}{c|c}
    \Theta & B 
    \end{array}
    \right]      
    \right)
    \begin{bmatrix}
        \tilde{x}_o \\ \eta 
    \end{bmatrix} =     
    \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} \tilde{x}_o 
    + B \eta 
    + \Theta \tilde{x}_o
\end{align*}
主系统变换推导
\begin{align*}
    \begin{bmatrix}
        \dot{\tilde{x}}_o \\ \dot{\eta} 
    \end{bmatrix}
    =    \begin{bmatrix}
        A_0 & b_0c_c \\
        0 & R(\theta)
    \end{bmatrix}
    \begin{bmatrix}
        \tilde{x}_o \\ \eta 
    \end{bmatrix}
\end{align*}
\begin{align*}
    \dot{\zeta} 
       = & \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} \dot{\tilde{x}}_o + B\dot{\eta} + \Theta \dot{\tilde{x}}_o
    \\ = & \begin{bmatrix}I_{v} \\ 0 \end{bmatrix}(A_o \tilde{x}_o + b_0 c_c \eta) + B R(\theta) \eta + \Theta \dot{\tilde{x}}_o
    \\ = & \begin{bmatrix}I_{v} \\ 0 \end{bmatrix}A_o \tilde{x}_o 
    + \left(\begin{bmatrix}I_{v} \\ 0 \end{bmatrix}b_0 c_c  + B R(\theta) \right)\eta 
    + \Theta \dot{\tilde{x}}_o
    \\ = & \left(-\begin{bmatrix}I_{v} \\ 0 \end{bmatrix}a_0e_1^\top + \begin{bmatrix}I_{v} \\ 0 \end{bmatrix}A_{cv} \right)\tilde{x}_o
    + \left(\begin{bmatrix}I_{v} \\ 0 \end{bmatrix}b_0 c_c  + B R(\theta) \right)\eta + \Theta \dot{\tilde{x}}_o
    \\ = & \left( -a[0]e_1^\top + \begin{bmatrix}I_{v} \\ 0 \end{bmatrix}A_{cv} \right)\tilde{x}_o
    + \left(\begin{bmatrix}I_{v} \\ 0 \end{bmatrix}b_0 c_c  
    + B \left(-\theta_0 e_1^\top + A_{cm}\right) \right)\eta 
    + \Theta \dot{\tilde{x}}_o
    \\ = & \left( -a[0]e_1^\top + \begin{bmatrix}I_{v} \\ 0 \end{bmatrix}A_{cv} \right)\tilde{x}_o
    + \left(\begin{bmatrix}I_{v} \\ 0 \end{bmatrix}b_0 c_c  
    + B A_{cm} -B\theta_0 e_1^\top \right)\eta 
    + \left( - \Theta a_0 e_1^\top\tilde{x}_o + \Theta A_{cv}\tilde{x}_o + \Theta b_oe_1^\top\eta\right)
\end{align*}
\begin{align*}
    \dot{\zeta} 
        = & A_c\zeta - a[0] e - \sum_{i=1}^m \theta_i a[i]e
    \\ = & 
    \left(\begin{bmatrix}
        A_{cv} & 0 \\
        0 & A_{cm}
    \end{bmatrix}
    + e_ve_{v+1}^\top\right)
    \left(\begin{bmatrix}I_{v} \\ 0 \end{bmatrix} \tilde{x}_o + B\eta + \Theta \tilde{x}_o\right) 
    - a[0] e - \sum_{i=1}^m \theta_i a[i]e
    \\ = &
    \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} A_{cv}\tilde{x}_o
     + A_c \left(B\eta + \Theta \tilde{x}_o\right) 
     - a[0] e - \sum_{i=1}^m \theta_i a[i]e
    \\ = &
    \left( - a[0] e + \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} A_{cv}\tilde{x}_o\right)
    + A_cB\eta 
    + \left(- \sum_{i=1}^m \theta_i a[i]e + A_c \Theta {\tilde{x}}_o \right)
    \\ = &
    \left( -a[0]e_1^\top + \begin{bmatrix}I_{v} \\ 0 \end{bmatrix}A_{cv} \right)\tilde{x}_o
    + \left( \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} b_0c_c + BA_{cm}\right)\eta 
    + \left[- \sum_{i=1}^m \theta_i a[i]e
    + \left(\begin{bmatrix}I_{2m} \\ 0 \end{bmatrix}\theta_0e_1^\top + \Theta A_{cv}\right)
    {\tilde{x}}_o \right]
    \\ = &
    \left( -a[0]e_1^\top + \begin{bmatrix}I_{v} \\ 0 \end{bmatrix}A_{cv} \right)\tilde{x}_o
    + \left( \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} b_0c_c + BA_{cm}\right)\eta 
    + \left(- \Theta a_0 e_1^\top\tilde{x}_o 
    + \Theta A_{cv}{\tilde{x}}_o \right)
\end{align*}
等效向量
\begin{align*}
    B\theta_0e_1^\top \eta 
    = & \left(\sum_{t=1}^{2m+1} Z_n^{t} \begin{bmatrix}I_{2m+1}\\ 0\end{bmatrix} b_0 e_{t}^\top \right)
    \theta_0 e_1^\top \eta \in \mathbb{R}^{n} 
    \\ = & 
    \left(\sum_{t=1}^{v} b_{v-t}\cdot Z_n^{t} \begin{bmatrix}I_{2m+1}\\ 0\end{bmatrix} \right)
    \left(\sum_{\ell=1}^{m} \theta_\ell e_{2\ell} \right)e_1^\top \eta
    \\ = & 
    \left(\sum_{t=1}^{v} \sum_{\ell=1}^{m} 
    b_{v-t}\theta_\ell\cdot Z_n^{t} \begin{bmatrix}I_{2m+1}\\ 0\end{bmatrix} e_{2\ell} \right) e_1^\top \eta
    \\ = & 
    \left(\sum_{t=1}^{v} \sum_{\ell=1}^{m} 
    b_{v-t}\theta_\ell\cdot Z_n^{t} e_{2\ell} \right) e_1^\top \eta
    \\ = & 
    \left(\sum_{t=1}^{v} \sum_{\ell=1}^{m} 
    b_{v-t}\theta_\ell\cdot e_{2\ell+t} \right) e_1^\top \eta
    =  \left( \sum_{\ell=1}^{m} \sum_{t=1}^{v}
    b_{v-t}\theta_\ell\cdot e_{2\ell+t} \right) e_1^\top \eta
    \\ = & 
    \left( \sum_{\ell=1}^{m} \sum_{t=1}^{v}
    b_{v-t}\theta_\ell\cdot Z_n^{2\ell} e_{t} \right) e_1^\top \eta
    \\ = & 
    \left( \sum_{\ell=1}^{m} 
    \theta_\ell Z_n^{2\ell} \sum_{t=1}^{v} b_{v-t} e_{t} \right) e_1^\top \eta
    \\ = & 
    \left( \sum_{\ell=1}^{m} \theta_\ell Z_n^{2\ell} \begin{bmatrix}I_{v}\\ 0\end{bmatrix}\right)
    \left(\sum_{t=1}^{v} b_{v-t} e_{t} \right) e_1^\top \eta
    \\ = & \Theta_0b_0 e_1^\top \eta 
\end{align*}
拓展系统变换推导
\begin{align*}
    \begin{bmatrix}
        \dot{\tilde{x}}_o \\ \dot{\eta} 
    \end{bmatrix}
    =    \begin{bmatrix}
        A_0 & b_0c_c \\
        0 & R(\theta)
    \end{bmatrix}
    \begin{bmatrix}
        \tilde{x}_o \\ \eta 
    \end{bmatrix} + 
    \begin{bmatrix}
        b_0 \\ 0
    \end{bmatrix} (u-\varepsilon)
\end{align*}
\begin{align*}
\dot{\zeta} 
       = & \begin{bmatrix}I_{v} \\ 0 \end{bmatrix} \dot{\tilde{x}}_o + B\dot{\eta} + \Theta \dot{\tilde{x}}_o
    \\ = & \left(\begin{bmatrix}I_{v} \\ 0 \end{bmatrix} + \Theta\right) (A_o \tilde{x}_o + b_0 c_c \eta + b_0(u-\epsilon )) + B R(\theta) \eta
    \\ = & A_c\zeta - a[0] e - \sum_{i=1}^m \theta_i a[i]e + \left(\begin{bmatrix}I_{v} \\ 0 \end{bmatrix} + \Theta\right)b_0(u-\epsilon )
    \\ = & A_c\zeta - a[0] e - \sum_{i=1}^m \theta_i a[i]e + b[0](u-\varepsilon) 
    + \left(\sum_{\ell=1}^{m} \sum_{t=1}^{v} b_{v-t}\theta_\ell\cdot e_{2\ell+t} \right) (u-\varepsilon) 
    \\ = & A_c\zeta - a[0] e - \sum_{i=1}^m \theta_i a[i]e + b[0](u-\varepsilon) 
    + \left( \sum_{\ell=1}^{m} \theta_\ell Z_n^{2\ell} \sum_{t=1}^{v} b_{v-t} e_{t} \right) (u-\varepsilon) 
    \\ = & A_c\zeta - a[0] e - \sum_{i=1}^m \theta_i a[i]e + b[0](u-\varepsilon) 
    + \left( \sum_{\ell=1}^{m} \theta_\ell Z_n^{2\ell} b_0 \right) (u-\varepsilon) 
    \\ = & A_c\zeta - a[0] e - \sum_{i=1}^m \theta_i a[i]e + b[0](u-\varepsilon) 
    +  \sum_{i=1}^{m} \theta_i b[i] (u-\varepsilon) 
\end{align*}
